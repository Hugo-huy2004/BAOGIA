import { useMemo, useState } from "react";
import { useTranslation } from "react-i18next";
import { joyCode, joyFactor } from "../../../lib/joyDisplay";
import { SEGMENT_SEC } from "../../../../shared/stockPricing";
import { CANDLE_FRAMES, candlesFromHistory, candlesFromTicks, formingCandle, movingAverage, volumeOf } from "./candles";

/**
 * Biểu đồ nến của sàn ảo — dựng đúng bộ phận của một biểu đồ chứng khoán thật:
 * thân nến, bóng trên/dưới, cột khối lượng, hai đường trung bình động.
 *
 * NẾN LẤY TỪ ĐƯỜNG GIÁ THẬT, KHÔNG VẼ THÊM GÌ. Máy chủ gửi xuống `ticks` — mỗi
 * mốc một phút, chính mảng nó dùng để khớp lệnh. Gộp N mốc thành một nến thì:
 *   mở  = mốc đầu · đóng = mốc cuối · cao/thấp = max/min các mốc trong nến
 * Giữa hai mốc giá đi thẳng (đó là cách priceAt nội suy), nên đỉnh và đáy thật
 * của đoạn đó LUÔN rơi vào một mốc — bóng nến vì thế là bóng thật, không phải
 * số bịa cho đẹp. Đây cũng là lý do khung nhỏ nhất là 2 phút chứ không phải 1:
 * một mốc thì mở = đóng = cao = thấp, vẽ ra chỉ là một gạch ngang.
 *
 * KHỐI LƯỢNG LÀ SỐ MÔ PHỎNG và nhãn ghi đúng như vậy. Sàn ảo chưa có sổ lệnh,
 * nên cột khối lượng suy từ biên độ nến (thị trường thật: giá càng giật thì
 * khối lượng càng lớn) cộng một nhiễu cố định theo mốc thời gian của nến — cố
 * định để cột không nhảy múa mỗi lần vẽ lại.
 *
 * ponytail: khối lượng mô phỏng. Khi nào máy chủ gộp được số lượng khớp thật
 * từ StockTrade theo từng nến thì thay vào đây, bỏ chữ "mô phỏng" ở chú giải.
 */


// Màu lấy từ token theme, KHÔNG hex cứng: biểu đồ phải đọc được ở cả nền sáng
// lẫn nền tối. Xanh/đỏ ở đây là NGHĨA tăng/giảm chứ không phải màu trang trí,
// nên dùng --success/--destructive chứ không phải màu nhấn của app.
const UP = "hsl(var(--success))";
const DOWN = "hsl(var(--destructive))";
const GRID = "hsl(var(--border))";
const MUTED = "hsl(var(--muted-foreground))";
// MA10 dùng đúng một màu nhấn của app; MA30 để trung tính — hai đường vẫn phân
// biệt được mà không kéo thêm màu thứ hai vào bảng màu.
const MA_FAST = "hsl(var(--primary))";
const MA_SLOW = "hsl(var(--muted-foreground))";

const axisText = (joy, locale) => {
  const shown = Number(joy || 0) * joyFactor();
  return shown.toLocaleString(locale, { maximumFractionDigits: Math.abs(shown) >= 1000 ? 0 : 2 });
};
const priceText = (joy, locale) => `${axisText(joy, locale)} ${joyCode()}`;
const timeText = (ms, withDate, locale) =>
  new Date(ms).toLocaleString(locale, withDate
    ? { day: "2-digit", month: "2-digit", hour: "2-digit", minute: "2-digit" }
    : { hour: "2-digit", minute: "2-digit" });

export default function StockPriceChart({ company, trades = [] }) {
  const { t, i18n } = useTranslation();
  const currentLocale = i18n.language || "vi-VN";
  const [timeframe, setTimeframe] = useState("2m");
  const [scrubIndex, setScrubIndex] = useState(null);

  const frame = CANDLE_FRAMES.find((item) => item.key === timeframe) || CANDLE_FRAMES[0];

  // `company.price` là giá nội suy, đổi MỖI GIÂY ở component cha — nó nằm trong
  // deps để cây nến bên phải lớn dần theo thời gian thực. Thiếu nó thì biểu đồ
  // chỉ nhúc nhích mỗi 30 giây, lúc máy chủ gửi mốc mới, nhìn như treo.
  const MAX_CANDLES = 30;
  const candles = useMemo(() => {
    let result = [];
    if (!frame.perCandle) {
      result = candlesFromHistory(company?.history);
    } else {
      const closed = candlesFromTicks(company?.ticks, frame.perCandle, SEGMENT_SEC);
      const forming = formingCandle(company?.ticks, frame.perCandle, company?.price, SEGMENT_SEC);
      result = forming ? [...closed, forming] : closed;
    }
    // Chỉ lấy 30 nến gần nhất để biểu đồ không bị nén chặt và không tính High/Low của dữ liệu khuất
    return result.slice(-MAX_CANDLES);
  }, [company?.ticks, company?.history, company?.price, frame.perCandle]);

  const closes = useMemo(() => candles.map((c) => c.close), [candles]);
  const ma10 = useMemo(() => movingAverage(closes, 10), [closes]);
  const ma30 = useMemo(() => movingAverage(closes, 30), [closes]);
  const volumes = useMemo(() => candles.map(volumeOf), [candles]);

  // ── Khung vẽ ────────────────────────────────────────────────────────────────
  const width = 480;
  const gutter = 62;          // chỗ cho thang giá bên phải, như mọi app chứng khoán
  const padX = 8;
  const priceTop = 12;
  const priceBottom = 186;
  const volTop = 200;
  const volBottom = 246;
  const height = 262;
  const plotWidth = width - gutter - padX;

  const high = candles.length ? Math.max(...candles.map((c) => c.high)) : 0;
  const low = candles.length ? Math.min(...candles.map((c) => c.low)) : 0;
  const span = high - low || 1;
  const maxVolume = volumes.length ? Math.max(...volumes) : 1;

  // Giữ nguyên slot = plotWidth / MAX_CANDLES để độ rộng nến luôn cố định dù chưa đủ 30 nến
  const slot = plotWidth / MAX_CANDLES;
  const bodyWidth = Math.max(1.5, Math.min(14, slot * 0.62));
  // Đẩy nến sang phải để nến mới nhất luôn bám biên phải
  const offset = MAX_CANDLES - candles.length;
  const xOf = (index) => padX + (index + offset + 0.5) * slot;
  const yOf = (price) => priceBottom - ((price - low) / span) * (priceBottom - priceTop);

  const maPath = (series) => {
    let path = "";
    let open = false;
    series.forEach((value, index) => {
      if (value === null) { open = false; return; }
      path += `${open ? "L" : "M"} ${xOf(index).toFixed(1)} ${yOf(value).toFixed(1)} `;
      open = true;
    });
    return path.trim();
  };

  const gridLines = useMemo(() => (
    candles.length ? [0, 0.25, 0.5, 0.75, 1].map((ratio) => ({
      y: priceTop + ratio * (priceBottom - priceTop),
      price: high - ratio * span,
    })) : []
  ), [candles.length, high, span]);

  // Tối đa 6 chấm, tính thẳng khi vẽ: bọc useMemo ở đây chỉ tổ phải khai báo
  // xOf/yOf làm phụ thuộc trong khi chúng đổi mỗi lần render.
  const marks = (() => {
    if (!candles.length) return [];
    const first = candles[0].time;
    const last = candles[candles.length - 1].time + 1;
    return (trades || [])
      .filter((t) => t.symbol === company?.symbol)
      .slice(0, 6)
      .map((trade) => {
        const at = new Date(trade.at || Date.now()).getTime();
        if (at < first || at > last) return null;
        const index = Math.min(candles.length - 1, Math.round(((at - first) / (last - first)) * (candles.length - 1)));
        return { ...trade, x: xOf(index), y: yOf(trade.price) };
      })
      .filter(Boolean);
  })();

  const active = scrubIndex !== null && candles[scrubIndex] ? candles[scrubIndex] : null;
  const latest = candles[candles.length - 1] || null;
  const shown = active || latest;
  const shownUp = shown ? shown.close >= shown.open : true;

  const handleMove = (event) => {
    const rect = event.currentTarget.getBoundingClientRect();
    const x = ((event.clientX ?? event.touches?.[0]?.clientX ?? 0) - rect.left) / rect.width * width;
    const visualIndex = Math.floor((x - padX) / slot);
    const index = visualIndex - offset;
    setScrubIndex(index >= 0 && index < candles.length ? index : null);
  };

  return (
    <div className="mt-3 min-w-0 space-y-2 overflow-hidden rounded-2xl border border-border bg-card p-3">
      <div className="flex min-w-0 flex-wrap items-center justify-between gap-2">
        {shown ? (
          <div className="flex min-w-0 flex-wrap items-center gap-x-2.5 gap-y-1 text-[12px] font-bold tabular-nums">
            <span className="text-muted-foreground">{timeText(shown.time, !frame.perCandle, currentLocale, currentLocale)}</span>
            <span className="text-muted-foreground">{t("invest.ui.openShort", "M")} <span className="text-foreground">{axisText(shown.open, currentLocale)}</span></span>
            <span className="text-muted-foreground">{t("invest.ui.highShort", "C")} <span className="text-foreground">{axisText(shown.high, currentLocale)}</span></span>
            <span className="text-muted-foreground">{t("invest.ui.lowShort", "T")} <span className="text-foreground">{axisText(shown.low, currentLocale)}</span></span>
            <span className="text-muted-foreground">{t("invest.ui.closeShort", "Đ")} <span style={{ color: shownUp ? UP : DOWN }}>{axisText(shown.close, currentLocale)}</span></span>
          </div>
        ) : <span className="text-xs font-bold text-muted-foreground">{t("invest.ui.notEnoughData", "Chưa đủ dữ liệu để vẽ nến")}</span>}

        <div className="flex gap-1 rounded-lg border border-border bg-muted p-1">
          {CANDLE_FRAMES.map((item) => (
            <button
              key={item.key}
              type="button"
              onClick={() => { setTimeframe(item.key); setScrubIndex(null); }}
              className={`min-h-9 rounded-md px-3 text-xs font-bold ${
                timeframe === item.key ? "bg-primary text-primary-foreground" : "text-muted-foreground"
              }`}
            >
              {t(`invest.ui.${item.key}`, item.label)}
            </button>
          ))}
        </div>
      </div>

      <div
        className="relative touch-none"
        onMouseMove={handleMove}
        onMouseLeave={() => setScrubIndex(null)}
        onTouchMove={handleMove}
        onTouchEnd={() => setScrubIndex(null)}
      >
        <svg viewBox={`0 0 ${width} ${height}`} className="w-full">
          {/* Lưới giá + thang giá bên phải */}
          {gridLines.map((line) => (
            <g key={line.y}>
              <line x1={padX} y1={line.y} x2={width - gutter} y2={line.y} stroke={GRID} strokeWidth="1" />
              <text x={width - gutter + 6} y={line.y + 3.5} fill={MUTED} className="text-[13px] font-bold tabular-nums">
                {axisText(line.price, currentLocale)}
              </text>
            </g>
          ))}

          {/* Đường trung bình động: MA10 nhanh, MA30 chậm — chỗ hai đường cắt nhau
              là chỗ bài học "xu hướng đổi chiều" nói tới. */}
          <path d={maPath(ma30)} fill="none" stroke={MA_SLOW} strokeWidth="1.3" />
          <path d={maPath(ma10)} fill="none" stroke={MA_FAST} strokeWidth="1.6" />

          {/* Nến */}
          {candles.map((candle, index) => {
            const up = candle.close >= candle.open;
            const color = up ? UP : DOWN;
            const x = xOf(index);
            const yOpen = yOf(candle.open);
            const yClose = yOf(candle.close);
            const top = Math.min(yOpen, yClose);
            // Nến không đổi giá vẫn phải thấy được: cho thân dày tối thiểu 1px.
            const bodyHeight = Math.max(1, Math.abs(yClose - yOpen));

            return (
              <g key={candle.time}>
                {!candle.noWick && (
                  <line x1={x} y1={yOf(candle.high)} x2={x} y2={yOf(candle.low)} stroke={color} strokeWidth="1.1" />
                )}
                <rect
                  x={x - bodyWidth / 2}
                  y={top}
                  width={bodyWidth}
                  height={bodyHeight}
                  fill={color}
                  rx={bodyWidth > 4 ? 1 : 0}
                />
              </g>
            );
          })}

          {/* Giá mới nhất: đường đứt + nhãn, đúng kiểu bảng điện */}
          {latest && (
            <g>
              <line
                x1={padX} y1={yOf(latest.close)} x2={width - gutter} y2={yOf(latest.close)}
                stroke={latest.close >= latest.open ? UP : DOWN} strokeDasharray="3 3" strokeWidth="1" strokeOpacity="0.7"
              />
              <rect
                x={width - gutter + 1} y={yOf(latest.close) - 9} width={gutter - 3} height="18" rx="3"
                fill={latest.close >= latest.open ? UP : DOWN}
              />
              <text x={width - gutter / 2 - 1} y={yOf(latest.close) + 4} textAnchor="middle" fill={latest.close >= latest.open ? "hsl(var(--success-foreground))" : "hsl(var(--destructive-foreground))"} className="text-[13px] font-bold tabular-nums">
                {axisText(latest.close, currentLocale)}
              </text>
            </g>
          )}

          {/* Lệnh của chính mình đã khớp ở đâu trên biểu đồ */}
          {marks.map((mark, index) => (
            <g key={`${mark.at || index}-${index}`}>
              <circle cx={mark.x} cy={mark.y} r="4.5" fill={mark.side === "buy" ? UP : DOWN} stroke="hsl(var(--card))" strokeWidth="1.6" />
              <text x={mark.x} y={mark.y - 9} textAnchor="middle" fill={mark.side === "buy" ? UP : DOWN} className="text-[13px] font-bold">
                {mark.side === "buy" ? t("invest.ui.buy", "MUA") : t("invest.ui.sell", "BÁN")}
              </text>
            </g>
          ))}

          {/* Cột khối lượng */}
          {candles.map((candle, index) => {
            const barHeight = Math.max(1, (volumes[index] / maxVolume) * (volBottom - volTop));
            return (
              <rect
                key={`v-${candle.time}`}
                x={xOf(index) - bodyWidth / 2}
                y={volBottom - barHeight}
                width={bodyWidth}
                height={barHeight}
                fill={candle.close >= candle.open ? UP : DOWN}
                fillOpacity="0.55"
              />
            );
          })}
          <line x1={padX} y1={volBottom} x2={width - gutter} y2={volBottom} stroke={GRID} strokeWidth="1" />

          {/* Vạch dò khi rê tay */}
          {active && (
            <line
              x1={xOf(scrubIndex)} y1={priceTop} x2={xOf(scrubIndex)} y2={volBottom}
              stroke="hsl(var(--foreground))" strokeDasharray="2 2" strokeWidth="1" strokeOpacity="0.5"
            />
          )}
        </svg>
      </div>

      <div className="flex flex-wrap items-center justify-between gap-x-3 gap-y-1 border-t border-border pt-2 text-xs font-bold text-muted-foreground">
        <div className="flex items-center gap-3">
          <span className="flex items-center gap-1"><span className="h-0.5 w-3 rounded-full bg-primary" /> MA10</span>
          <span className="flex items-center gap-1"><span className="h-0.5 w-3 rounded-full bg-muted-foreground" /> MA30</span>
          <span>{t("invest.ui.simulatedVol", "KL mô phỏng")}</span>
        </div>
        <span className="text-foreground">
          {candles.length ? `${candles.length} ${t("invest.ui.candles", "nến")} · ${t(`invest.ui.${frame.key}`, frame.label)}` : "—"}
          {latest ? ` · ${priceText(latest.close, currentLocale)}` : ""}
        </span>
      </div>
    </div>
  );
}
